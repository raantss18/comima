% Traduction française COMIMa de l'énoncé officiel anglais.
Soit $n$ un entier fixé, avec $n \geq 2$.
\begin{enumerate}
\item Déterminer la plus petite constante $C$ telle que l'inégalité
\[
\sum_{1 \leq i < j \leq n} x_i x_j (x_i^2 + x_j^2)
\leq C \left(\sum_{1 \leq i \leq n} x_i\right)^4
\]
soit vérifiée pour tous les réels $x_1, \ldots, x_n \geq 0$.
\item Pour cette constante $C$, déterminer les cas d'égalité.
\end{enumerate}
