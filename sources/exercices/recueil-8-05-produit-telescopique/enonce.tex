On pose
\[
A = \frac12 \cdot \frac34 \cdot \frac56 \cdots \frac{99}{100}
= \prod_{k=1}^{50} \frac{2k-1}{2k}.
\]
On veut démontrer que $\dfrac{1}{15} < A < \dfrac{1}{10}$.
\begin{enumerate}
  \item Montrer que, pour tout entier $k \geqslant 1$,
  \[
  \frac{2k-1}{2k} < \frac{2k}{2k+1}
  \qquad\text{et}\qquad
  \frac{2k+1}{2k+2} > \frac{2k}{2k+1}.
  \]
  \item \emph{Majoration.} On pose
  $B = \dfrac23 \cdot \dfrac45 \cdots \dfrac{100}{101}$. Montrer que
  $A < B$, calculer le produit $AB$ (il est télescopique) et en déduire que
  $A < \dfrac{1}{10}$.
  \item \emph{Minoration.} On pose
  $C = \dfrac12 \cdot \dfrac23 \cdot \dfrac45 \cdots \dfrac{98}{99}$.
  Montrer que $A > C$, calculer $AC$ et en déduire que $A > \dfrac{1}{15}$.
\end{enumerate}
