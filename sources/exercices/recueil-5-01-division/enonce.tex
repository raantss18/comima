\begin{itemize}
  \item Trouver tous les entiers $n \geq 1$ tels que $\dfrac{n^3 + 3}{n^2 + 7}$
  soit un entier.
  \item Soient $a$ et $b$ deux entiers et $n \geq 1$ un entier. Montrer que $a - b$
  divise $a^n - b^n$. Montrer que, si $n$ est impair, $a + b$ divise $a^n + b^n$
  (si $a=b$ ou $a=-b$, le diviseur est nul et l'on convient que $0$ divise
  seulement $0$).
\end{itemize}
