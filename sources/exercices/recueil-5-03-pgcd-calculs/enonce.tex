\begin{itemize}
  \item Soit $n$ un entier. Calculer $\mathrm{PGCD}(5n+2,\ 12n+5)$ et
  $\mathrm{PGCD}(9n+4,\ 2n-1)$ (la réponse peut dépendre de $n$).
  \item Soient $a$ et $b$ deux entiers non tous nuls. Soient $c, x, y$
  trois entiers tels que $c = ax + by$. Montrer que $\mathrm{PGCD}(a, b)$
  divise $c$.
\end{itemize}
