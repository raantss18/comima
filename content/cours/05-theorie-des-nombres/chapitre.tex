\chapter{Théorie des nombres}


\section{Division euclidienne}

\begin{thm}
	{Théorème — Division euclidienne}\\
	Soient $a, b \in \mathbb{Z}$, $b \neq 0$. Il existe un unique couple $(q, r) \in \mathbb{Z} \times \mathbb{N}$ tel que
	\[
	a = bq + r, \qquad 0 \leq r \leq |b| - 1.
	\]
	L'entier naturel $r$ ainsi obtenu est appelé \textit{le reste de la division euclidienne} de $a$ par $b.$
\end{thm}

\begin{itemize}
	\item \textbf{Divisibilité :} $b$ divise $a$ s'il existe un entier $b$ tel que $a = bq$, i.e. le reste est $0.$
\end{itemize}

\subsection*{Nombres premiers}

\begin{defi}
	Soit $p \geq 2$ un entier. Le nombre entier $p$ est \textit{premier} si ses seuls diviseurs positifs sont 1 et $p$.\\
	\quad Un entier $n \geq 2$ est dit \textit{composé} (non premier) si $n = ab$ avec $a \geq 2$ et $b \geq 2$.
\end{defi}

\begin{fact}
	Tout entier $n \geq 2$ admet un diviseur premier.
\end{fact}

\begin{fact}
	Si $p$ premier, $p \mid ab$ alors $p \mid a$ ou $p \mid b$.
\end{fact}

\begin{fact}{Théorème fondamental de l'arithmétique}\\
	Tout entier supérieur ou égal à $2$ peut être représenté comme produit de facteurs de nombres premiers,
	de manière unique.
\end{fact}

\begin{exercise}
	\begin{itemize}
		\item[$\bullet$] Montrer qu'un nombre premier supérieur à $3$ est de la forme $6n \pm 1$.
		\item[$\bullet$]Soient $a$ et $b$ deux entiers. Montrer que $a^4 + 4b^4$ n'est pas un nombre premier (pour $a, b > 1$).
	\end{itemize}
\end{exercise}

\section{PGCD et PPCM}

D'après les \textbf{Fait 1}, \textbf{Fait 3} et l'un des principes extrémaux, les définitions suivantes ont un sens:
\begin{defi}
	Soient $a$ et $b$ deux entiers tels que $(a, b) \neq (0, 0)$. Le plus grand des diviseurs positifs communs à $a$ et $b$ existe et sera noté par $\pgcd(a,b).$ Tandis que le plus petit des multiples communs à $a$ et $b$ sera noté par $\ppcm(a,b).$ En particulier, si $\pgcd(a,b)=1,$ on dit dans ce cas que $a$ et $b$ sont premiers entre eux.
\end{defi}

\begin{fact}
	$\pgcd(a, b) = \pgcd(b,\, a - b) = \pgcd(b,\, r)$ où $a = bq + r$.
\end{fact}

\begin{fact}
	$|ab| = \pgcd(a, b) \cdot \ppcm(a, b)$.
\end{fact}

\begin{fact}
	\textit{Théorème de Bézout}\\
	Soient $a$ et $b$ deux entiers non tous nuls. Alors, ils existent deux entiers $x$ et $y$ tels que $\pgcd(a, b) = ax + by$.\\
	\quad En particulier, si $a$ et $b$ sont premiers entre eux si et seulement si on a : $ax + by = 1$.
\end{fact}

\begin{exercise}
	\begin{itemize}
		\item[$\bullet$] Soient $n$ un entier. Calculer $\pgcd(5n + 2,\; 12n + 5)$ et $\pgcd(9n + 4,\; 2n - 1)$.\\
	\item[$\bullet$]Soient $a$ et $b$ deux entiers non tous nuls. Supposons qu'ils existent trois entiers $c, x$ et $y$ tels que $c = ax + by.$ Alors, on a: $\pgcd(a, b)$ divise $c.$
\end{itemize}
\end{exercise}

\section{Congruences}

\begin{defi}
	Soit $n \in \mathbb{N}^*.$ Considérons deux entiers $a$ et $b.$ On dit que $a$ et $b$ sont congrus modulo $n$ (noté $a \equiv b \pmod{n}$) si $a$ et $b$ admettent le même reste après division euclidienne par $n$.
\end{defi}

\begin{fact} Si $\pgcd(c, n) = 1$ et $ca \equiv cb \pmod{n}$, alors $a \equiv b \pmod{n}$.
\end{fact}

\begin{fact}
	Soit $f$ un polynôme à coefficients dans $\mathbb{Z}$. Si $a \equiv b \pmod{n}$, alors $f(a) \equiv f(b) \pmod{n}$.
\end{fact}

\begin{fact}
	\textit{Petit théorème de Fermat}\\
	Soient $a$ un entier et $p$ premier, on a :
	\[
	a^p \equiv a \pmod{p}.
	\]
\end{fact}

\begin{exercise}
	\begin{itemize}
		\item[$\bullet$] Montrer que l'équation $15x^2 - 7y^2 = 9$ n'admet pas de solutions entières.
		\item[$\bullet$] Montrer qu'il n'existe pas de polynôme à coefficients entiers tel que $f(7) = 11$ et
		$f(11) = 13$.
	\end{itemize}
\end{exercise}

\section{Partie entière}

D'après l'un des principes extrémaux, la définition suivante a un sens:
\begin{defi}
	Soit $x \in \mathbb{R}$. le plus petit entier inférieur ou égal à $x$ noté par $\lfloor x \rfloor$ ou $E(x),$ est appelé la partie entière de $x.$
\end{defi}

\begin{exercise} Soient $x, y$ deux nombres réels et $n$ un entier naturel non nul.
	\begin{itemize}
		\item[$\bullet$] Montrer : $E(x) + E(y) \leq E(x + y)$.
		\item[$\bullet$] Montrer : $\lfloor \lfloor x \rfloor / n \rfloor = \lfloor x/n \rfloor$.
		\item[$\bullet$] $\lfloor x \rfloor + \lfloor y \rfloor + \lfloor x + y \rfloor \leq \lfloor 2x \rfloor + \lfloor 2y \rfloor$.
		\item[$\bullet$] Calculer $\left\lfloor \sqrt{1} \right\rfloor + \left\lfloor \sqrt{2} \right\rfloor + \cdots + \left\lfloor \sqrt{2026} \right\rfloor$.
	\end{itemize}
\end{exercise}
